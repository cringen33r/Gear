    return () => {
        if (!hasValue) {

        return latest as T;
    };
}

function createUser(id: string, name: string): User {
    return { id, name, email: `${id}@mail.test`, phone: '', description: '' } as User;
}

function usersLoadedSignal(users: User[]) {
    return createSignal<UsersLoadedSignal>(UsersLoadedSignal, {
        correlationId: CORRELATION_ID,
        organizationId: ORGANIZATION_ID,
        users,
    });
}

/** n = 1 -> id '001', name 'user1', email '1@1.1', phone 'phone-001', description 'descr1' */
function userCreatedSignal(n: number) {
    return createSignal<UserCreatedSignal>(UserCreatedSignal, {
        id: `00${n}`,
        name: `user${n}`,
        email: `${n}@${n}.${n}`,
        phone: `phone-00${n}`,
        description: `descr${n}`,
        correlationId: CORRELATION_ID,
        organizationId: ORGANIZATION_ID,
        departmentId: '1',
    });
}

describe('UsersService: ', () => {
    let signals$: Subject<Signal>;
    let signaler: Mocked<SignalerService>;
    let usersApi: Mocked<UsersApiService>;

    function createSut(usersInStore: User[] = []) {
        const store = new SortedUsersStore();
        store.initialize(usersInStore, ORGANIZATION_ID);

        TestBed.configureTestingModule({

                { provide: ClientsApiService, useValue: mockInstance(ClientsApiService) },

        ]);
    });

        // arrange
        const usersInStore = Array.from({ length: 60000 }, (_, i) => {
            const x = 60000 - i;
            return createUser(`${x}`, `user${x}`);
